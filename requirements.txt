# =============================================================================
# Solutions Project Accelerator - Python runtime requirements (Windows, cp312).
#
# Install into a fresh Python 3.12 environment with:
#
#   python -m pip install --upgrade pip
#   python -m pip install numpy==1.26.4 cython==3.0.10 patch-ng "setuptools<81" wheel
#   python -m pip install -r requirements.txt --no-build-isolation
#
# The pre-install / --no-build-isolation dance is only needed because a few
# legacy sdists (py-lmdb, pyastar2d) still declare setup-time numpy/patch-ng.
#
# setuptools MUST be pinned below 81: newer releases dropped the vendored
# `pkg_resources` module, which PyInstaller's paddle/paddleocr hooks import
# at build time -> "ModuleNotFoundError: No module named 'pkg_resources'"
# when running `pyinstaller runme.spec`. If you already have an env with a
# newer setuptools, fix it with: python -m pip install "setuptools<81"
#
# The Potrace / PaddleOCR engine binaries are NOT in git; fetch
# them with build_scripts/setup_vendor_deps.ps1 AFTER installing this file
# (the PaddleOCR model URLs are read from the paddleocr package below).
#
# Sidecar (Rockwell PLC bridge) deps live in plc_bridge/requirements.txt and
# must be installed in a SEPARATE env - see build_scripts/build_all.ps1.
# =============================================================================
